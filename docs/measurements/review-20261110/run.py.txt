def check_empty(bots=0):
    status = q('status')
    occupancy = re.search(r'players\s*:\s*(\d+) humans, (\d+) bots', status)
    if not occupancy or int(occupancy[1]) != 0 or (bots is not None and int(occupancy[2]) != bots):
        raise RuntimeError('Unexpected occupancy: '+(occupancy[0] if occupancy else 'status unavailable')+'; no occupied restart authorized')
    return status

def loaded(wanted):
    return bool(re.search(r'(?m)^\s*\[#\d+:LOADED\]: "WeaponPaints" \('+re.escape(wanted)+r'\)', q('css_plugins list')))

def cold_install(data, wanted):
    check_empty()
    manager = '/mnt/ssd/cs2_7/cs2server'
    s.stat(manager)
    def manage(action):
        _, out, err = c.exec_command('su - '+shlex.quote(x['user'])+' -c '+shlex.quote(manager+' '+action), timeout=120)
        out.read(); err.read()
        if out.channel.recv_exit_status() != 0: raise RuntimeError('Scoped manager '+action+' failed')
    manage('stop')
    _, out, err = c.exec_command('pgrep -u cs2_7 -x cs2', timeout=15)
    if out.read().strip(): raise RuntimeError('Instance is still running; refusing file replacement')
    with s.open(remote, 'wb') as f: f.write(data)
    manage('start')
    for attempt in range(60):
        time.sleep(2)
        try:
            check_empty()
            if loaded(wanted): return
        except (OSError, EOFError, ConnectionError): pass
        except RuntimeError as error:
            if 'status unavailable' not in str(error): raise
    raise RuntimeError('Expected WeaponPaints did not load after scoped cold start')

def manifest():
    # Hash contents in the remote process, never return config values or process arguments.
    code = """import pathlib,hashlib,json
r=pathlib.Path('/mnt/ssd/cs2_7/serverfiles/game/csgo/addons/counterstrikesharp')
out={}
for sub in ['plugins','shared','configs','gamedata']:
 for p in sorted((r/sub).rglob('*')):
  if p.is_file() and p.suffix in ['.dll','.json','.so','.vxml','.vcss']:
   out[str(p.relative_to(r))]=hashlib.sha256(p.read_bytes()).hexdigest()
print(json.dumps(out))
"""
    _, out, err = c.exec_command("python3 - <<'PY'\n"+code+"\nPY", timeout=60)
    result = out.read().decode(); error = err.read().decode()
    if error: raise RuntimeError('Remote hash collection failed')
    return json.loads(result)

    for suffix in ['.dll', '.deps.json']:

    q('css_plugins load '+folder)

def read_new_logs(name):

        with s.open(path, 'rb') as f:
            f.seek(log_offsets.get(filename, 0))
            result += f.read().decode('utf-8', errors='replace')
    return result
